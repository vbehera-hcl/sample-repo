# AWS CLI Commands to Create Nautilus EC2 Web Server with Nginx
# ================================================================

# Step 1: Create a Security Group that allows HTTP traffic on port 80
aws ec2 create-security-group \
    --group-name nautilus-web-sg \
    --description "Security group for Nautilus web server allowing HTTP traffic"

# Step 2: Add inbound rule to allow HTTP traffic on port 80 from anywhere
aws ec2 authorize-security-group-ingress \
    --group-name nautilus-web-sg \
    --protocol tcp \
    --port 80 \
    --cidr 0.0.0.0/0

# Step 3: Create the EC2 instance with User Data script to install and start Nginx
# Note: Replace ami-0ubuntu-id with an actual Ubuntu AMI ID for your region
# Common Ubuntu AMIs:
#   - us-east-1: ami-0c7217cdde317cfec (Ubuntu 22.04 LTS)
#   - us-west-2: ami-03f65b8614a860c29 (Ubuntu 22.04 LTS)

aws ec2 run-instances \
    --image-id ami-0c7217cdde317cfec \
    --instance-type t2.micro \
    --security-groups nautilus-web-sg \
    --tag-specifications 'ResourceType=instance,Tags=[{Key=Name,Value=nautilus-ec2}]' \
    --user-data '#!/bin/bash
apt-get update -y
apt-get install -y nginx
systemctl start nginx
systemctl enable nginx' \
    --count 1
